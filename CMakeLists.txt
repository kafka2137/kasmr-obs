cmake_minimum_required(VERSION 3.28)
project(kasmr-obs VERSION 1.1.0 LANGUAGES C CXX)
file(WRITE "${CMAKE_BINARY_DIR}/project-version.txt" "${PROJECT_VERSION}\n")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(KASMR_BUILD_PLUGIN "Build Windows OBS source" ON)
include(CTest)
add_library(kasmr-core STATIC src/audio.cpp src/keyboard.cpp)
target_include_directories(kasmr-core PUBLIC src PRIVATE third_party)
if(MSVC)
  target_compile_options(kasmr-core PRIVATE /W4 /EHsc)
endif()
if(KASMR_BUILD_PLUGIN)
  if(NOT WIN32)
    message(FATAL_ERROR "The OBS plugin currently supports Windows only")
  endif()
  find_package(libobs QUIET CONFIG)
  if(NOT TARGET OBS::libobs)
    include(cmake/InstalledOBS.cmake)
  endif()
  add_library(kasmr-obs MODULE src/plugin.cpp src/windows-input.cpp)
  target_compile_definitions(kasmr-obs PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(kasmr-obs PRIVATE kasmr-core OBS::libobs user32)
  set_target_properties(kasmr-obs PROPERTIES PREFIX "")
  install(TARGETS kasmr-obs RUNTIME DESTINATION kasmr-obs/bin/64bit LIBRARY DESTINATION kasmr-obs/bin/64bit)
  install(DIRECTORY data/ DESTINATION kasmr-obs/data)
  install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION kasmr-obs/data)
  file(TO_NATIVE_PATH "${CMAKE_SOURCE_DIR}" INSTALLER_SOURCE_DIR)
  file(TO_NATIVE_PATH "${CMAKE_SOURCE_DIR}/dist/kasmr-obs" INSTALLER_STAGE_DIR)
  file(TO_NATIVE_PATH "${CMAKE_SOURCE_DIR}/dist" INSTALLER_OUTPUT_DIR)
  configure_file(cmake/windows/installer-Windows.iss.in installer-Windows.generated.iss @ONLY)
endif()
if(BUILD_TESTING)
  add_executable(core-tests tests/core-tests.cpp)
  target_link_libraries(core-tests PRIVATE kasmr-core)
  add_test(NAME core COMMAND core-tests "${CMAKE_CURRENT_SOURCE_DIR}/data/sounds" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/custom-pack")
  if(KASMR_BUILD_PLUGIN)
    add_executable(obs-smoke tests/obs-smoke.cpp)
    target_link_libraries(obs-smoke PRIVATE OBS::libobs)
    add_executable(installer-process tests/installer-process.cpp)
    set_target_properties(installer-process PROPERTIES OUTPUT_NAME obs64
      RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/installer-process")
  endif()
endif()
